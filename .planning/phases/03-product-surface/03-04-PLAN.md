```typescript
import { NextResponse } from 'next/server'
import { createClient } from '@/lib/supabase/server'
import { videoAnalysisQueue, frameGenerationQueue, videoSynthesisQueue, postProcessingQueue } from '@/lib/queues'

export async function GET() {
  // Admin auth check
  const supabase = await createClient()
  const { data: { user } } = await supabase.auth.getUser()
  if (!user || user.app_metadata?.role !== 'admin') {
    return NextResponse.json({ error: 'Forbidden' }, { status: 403 })
  }

  const queues = [videoAnalysisQueue, frameGenerationQueue, videoSynthesisQueue, postProcessingQueue]
  const health = await Promise.all(
    queues.map(async (q) => {
      const counts = await q.getJobCounts('active', 'waiting', 'failed', 'completed')
      return { name: q.name, ...counts }
    })
  )

  return NextResponse.json(health)
}
```

3. **Create `src/app/api/admin/retry-step/route.ts`** — retry failed step (ADMIN-03):

```typescript
import { NextResponse } from 'next/server'
import { createClient } from '@/lib/supabase/server'
import { videoAnalysisQueue, frameGenerationQueue, videoSynthesisQueue, postProcessingQueue } from '@/lib/queues'
import { z } from 'zod'

const retrySchema = z.object({
  jobId: z.string().uuid(),
  stepName: z.enum(['analysis', 'frame_generation', 'video_synthesis', 'post_processing']),
})

const STEP_TO_QUEUE: Record<string, typeof videoAnalysisQueue> = {
  analysis: videoAnalysisQueue,
  frame_generation: frameGenerationQueue,
  video_synthesis: videoSynthesisQueue,
  post_processing: postProcessingQueue,
}

export async function POST(req: Request) {
  const supabase = await createClient()
  const { data: { user } } = await supabase.auth.getUser()
  if (!user || user.app_metadata?.role !== 'admin') {
    return NextResponse.json({ error: 'Forbidden' }, { status: 403 })
  }

  const body = await req.json()
  const parsed = retrySchema.safeParse(body)
  if (!parsed.success) return NextResponse.json({ error: 'Invalid request' }, { status: 400 })

  const { jobId, stepName } = parsed.data

  // Reset step status to pending (ADMIN-03: does not restart prior steps)
  const { error: resetError } = await supabase
    .from('job_steps')
    .update({ status: 'pending', error: null, started_at: null, completed_at: null })
    .eq('job_id', jobId)
    .eq('step', stepName)

  if (resetError) return NextResponse.json({ error: resetError.message }, { status: 500 })

  // Re-enqueue to the specific queue
  const queue = STEP_TO_QUEUE[stepName]
  // Get job data for queue payload
  const { data: job } = await supabase
    .from('jobs')
    .select('id, org_id, product_name, reference_video_url')
    .eq('id', jobId)
    .single()

  if (!job) return NextResponse.json({ error: 'Job not found' }, { status: 404 })

  await queue.add('retry', {
    jobId: job.id,
    orgId: job.org_id,
    productName: job.product_name,
    referenceVideoUrl: job.reference_video_url,
    isRetry: true,
  }, { jobId: `retry-${jobId}-${stepName}-${Date.now()}` })

  return NextResponse.json({ ok: true })
}
```

4. **Create `src/app/api/admin/jobs/[jobId]/stage/route.ts`** — update SOP stage on drag (ADMIN-06):

```typescript
import { NextRequest, NextResponse } from 'next/server'
import { createClient } from '@/lib/supabase/server'
import { z } from 'zod'

const stageSchema = z.object({
  sop_stage: z.enum(['client', 'brief', 'generation', 'qc', 'delivered']),
})

export async function PATCH(
  req: NextRequest,
  { params }: { params: Promise<{ jobId: string }> }
) {
  const supabase = await createClient()
  const { data: { user } } = await supabase.auth.getUser()
  if (!user || user.app_metadata?.role !== 'admin') {
    return NextResponse.json({ error: 'Forbidden' }, { status: 403 })
  }

  const { jobId } = await params
  const body = await req.json()
  const parsed = stageSchema.safeParse(body)
  if (!parsed.success) return NextResponse.json({ error: 'Invalid stage' }, { status: 400 })

  const { error } = await supabase
    .from('jobs')
    .update({ sop_stage: parsed.data.sop_stage })
    .eq('id', jobId)

  if (error) return NextResponse.json({ error: error.message }, { status: 500 })
  return NextResponse.json({ ok: true })
}
```
