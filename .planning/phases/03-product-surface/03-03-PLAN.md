    </div>
  )
}

1. **Create `src/components/dashboard/JobProgressSSE.tsx`** — SSE progress display (CLI-02). Client component:

```tsx
'use client'
import { useEffect, useState } from 'react'

type StepStatus = { step: string; status: string; started_at?: string; completed_at?: string }

export function JobProgressSSE({ jobId }: { jobId: string }) {
  const [steps, setSteps] = useState<StepStatus[]>([])
  const [connected, setConnected] = useState(false)

  useEffect(() => {
    const es = new EventSource(`/api/status/${jobId}`)
    setConnected(true)

    es.onmessage = (e) => {
      const data = JSON.parse(e.data)
      if (data.type === 'snapshot') {
        setSteps(data.steps)
      } else if (data.type === 'update') {
        setSteps(prev => {
          const idx = prev.findIndex(s => s.step === data.step)
          if (idx >= 0) {
            const next = [...prev]
            next[idx] = { ...next[idx], ...data }
            return next
          }
          return [...prev, data]
        })
      }
    }

    es.addEventListener('done', () => { es.close(); setConnected(false) })
    es.onerror = () => { es.close(); setConnected(false) }

    // CRITICAL: cleanup on unmount to prevent connection leak (Pitfall 2)
    return () => es.close()
  }, [jobId])

  // Render 4 steps as a vertical timeline with status badges
  // Use DESIGN.md status badge colors: pending=gray, processing=amber, complete=emerald, failed=red
  return (/* step timeline UI */)
}
```
