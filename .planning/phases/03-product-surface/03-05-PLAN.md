```typescript
async function cleanupIntermediateFiles(jobId: string, supabase: SupabaseClient) {
  const prefixes = [
    `keyframes/${jobId}/`,
    `raw-clips/${jobId}/`,
  ]

  for (const prefix of prefixes) {
    const { data: files } = await supabase.storage
      .from('intermediates')
      .list(prefix)

    if (files?.length) {
      const paths = files.map(f => `${prefix}${f.name}`)
      await supabase.storage.from('intermediates').remove(paths)
    }
  }
}
```

```typescript
import { refundCredits } from '@/lib/credits/atomic'

// In the worker's failed event handler:
worker.on('failed', async (job, err) => {
  if (!job?.data?.jobId) return

  const { jobId, orgId } = job.data

  // Mark job as failed in DB
  await supabase.from('jobs').update({ status: 'failed' }).eq('id', jobId)

  // BILL-05: refund credits on confirmed failure
  const creditCost = job.data.creditCost ?? 1  // default cost per job
  await refundCredits(supabase, orgId, creditCost)
})
```

3. **Create `src/app/api/cron/stripe-reconciliation/route.ts`** — nightly reconciliation (QC-06):

```typescript
import { NextRequest, NextResponse } from 'next/server'
import { createClient } from '@supabase/supabase-js'

const supabase = createClient(
  process.env.NEXT_PUBLIC_SUPABASE_URL!,
  process.env.SUPABASE_SERVICE_ROLE_KEY!,
)

export async function GET(req: NextRequest) {
  // Verify cron secret
  if (req.headers.get('authorization') !== `Bearer ${process.env.CRON_SECRET}`) {
    return new Response('Unauthorized', { status: 401 })
  }

  // Sum credits from stripe_events (purchases)
  const { data: stripeData } = await supabase
    .from('credit_transactions')
    .select('delta')
    .eq('reason', 'purchase')

  const totalPurchased = (stripeData ?? []).reduce((sum, r) => sum + r.delta, 0)

  // Sum all credit deltas (purchases + deductions + refunds)
  const { data: allTx } = await supabase
    .from('credit_transactions')
    .select('delta')

  const netCredits = (allTx ?? []).reduce((sum, r) => sum + r.delta, 0)

  // Sum current org balances
  const { data: orgs } = await supabase
    .from('orgs')
    .select('credit_balance')

  const totalBalance = (orgs ?? []).reduce((sum, o) => sum + (o.credit_balance ?? 0), 0)

  // Reconcile: total balance should equal net credits
  const mismatch = Math.abs(totalBalance - netCredits)
  const ok = mismatch === 0

  const result = {
    totalPurchased,
    netCredits,
    totalBalance,
    mismatch,
    ok,
    checkedAt: new Date().toISOString(),
  }

  if (!ok) {
    console.error('[reconciliation] MISMATCH detected!', result)
    // TODO: send alert via email (Resend) when mismatch > threshold
  }

  return NextResponse.json(result)
}
```
